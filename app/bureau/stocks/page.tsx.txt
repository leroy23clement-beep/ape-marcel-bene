import { createClient } from "@/lib/supabase/server";
import { getInventoryItems, addItem, addPurchase } from "@/lib/actions/inventory";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { PackagePlus, ShoppingCart, Boxes } from "lucide-react";
import { redirect } from "next/navigation";
import Navbar from "@/components/Navbar";

export default async function StocksPage() {
  const supabase = await createClient();

  // Sécurité : vérifier l'utilisateur et son rôle CODIR
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) redirect("/login");

  const { data: profile } = await supabase
    .from("profiles")
    .select("*")
    .eq("id", user.id)
    .single();

  const allowedRoles = ["admin", "bureau", "tresorier", "secretaire"];
  const userRole = profile?.role ? profile.role.toLowerCase() : "parent";
  if (!allowedRoles.includes(userRole)) {
    redirect("/dashboard");
  }

  const items = await getInventoryItems();

  return (
    <div className="min-h-screen bg-gray-50 pb-12">
      <Navbar userEmail={user.email} firstName={profile?.first_name} role={profile?.role} />
      
      <main className="container mx-auto max-w-5xl space-y-8 p-6">
        <div className="flex flex-col gap-2 border-b pb-4">
          <h1 className="text-3xl font-bold tracking-tight text-gray-900">Gestion de la Réserve & Achats</h1>
          <p className="text-sm text-gray-600">
            Suivez les stocks globaux de l'association et enregistrez vos tickets d'achats fournisseurs.
          </p>
        </div>

        {/* Grille des formulaires d'action */}
        <div className="grid gap-6 md:grid-cols-2">
          
          {/* Formulaire : Nouvel Article */}
          <Card className="bg-white shadow-sm border">
            <CardHeader>
              <CardTitle className="flex items-center gap-2 text-lg text-gray-800">
                <PackagePlus className="size-5 text-amber-600" />
                Créer un nouvel article
              </CardTitle>
            </CardHeader>
            <CardContent>
              <form action={addItem} className="space-y-4">
                <div>
                  <Label htmlFor="name">Nom de l'article</Label>
                  <Input id="name" name="name" placeholder="ex: Coca-Cola 33cl" required className="mt-1" />
                </div>
                <div>
                  <Label htmlFor="category">Catégorie</Label>
                  <Input id="category" name="category" placeholder="ex: Buvette, Restauration" required defaultValue="Buvette" className="mt-1" />
                </div>
                <div>
                  <Label htmlFor="unit">Unité</Label>
                  <Input id="unit" name="unit" placeholder="ex: unité, pack, kg" required defaultValue="unité" className="mt-1" />
                </div>
                <Button type="submit" className="w-full bg-amber-600 hover:bg-amber-700 text-white">Ajouter au catalogue</Button>
              </form>
            </CardContent>
          </Card>

          {/* Formulaire : Enregistrer un Achat */}
          <Card className="bg-white shadow-sm border">
            <CardHeader>
              <CardTitle className="flex items-center gap-2 text-lg text-gray-800">
                <ShoppingCart className="size-5 text-amber-600" />
                Enregistrer un achat / réassort
              </CardTitle>
            </CardHeader>
            <CardContent>
              <form action={addPurchase} className="space-y-4">
                <div>
                  <Label htmlFor="item_id">Article</Label>
                  <select
                    id="item_id"
                    name="item_id"
                    className="w-full mt-1 rounded-md border border-input bg-background px-3 py-2 text-sm shadow-sm focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring"
                    required
                  >
                    <option value="">Sélectionner un article</option>
                    {items?.map((item: any) => (
                      <option key={item.id} value={item.id}>
                        {item.name} (Stock : {item.current_stock} {item.unit})
                      </option>
                    ))}
                  </select>
                </div>
                <div className="grid grid-cols-2 gap-4">
                  <div>
                    <Label htmlFor="quantity">Quantité achetée</Label>
                    <Input id="quantity" name="quantity" type="number" min="1" placeholder="ex: 48" required className="mt-1" />
                  </div>
                  <div>
                    <Label htmlFor="total_cost">Coût total (€)</Label>
                    <Input id="total_cost" name="total_cost" type="number" step="0.01" placeholder="ex: 35.50" required className="mt-1" />
                  </div>
                </div>
                <div>
                  <Label htmlFor="supplier">Fournisseur</Label>
                  <Input id="supplier" name="supplier" placeholder="ex: Métro, E.Leclerc" className="mt-1" />
                </div>
                <Button type="submit" className="w-full bg-amber-600 hover:bg-amber-700 text-white">Valider l'achat et réapprovisionner</Button>
              </form>
            </CardContent>
          </Card>
        </div>

        {/* Tableau de l'état actuel de la réserve */}
        <Card className="bg-white shadow-sm border">
          <CardHeader>
            <CardTitle className="flex items-center gap-2 text-xl text-gray-800">
              <Boxes className="size-6 text-amber-600" />
              État de la Réserve Actuelle
            </CardTitle>
          </CardHeader>
          <CardContent>
            <div className="overflow-x-auto">
              <table className="w-full text-left border-collapse">
                <thead>
                  <tr className="border-b text-sm text-gray-500">
                    <th className="pb-3 font-medium">Article</th>
                    <th className="pb-3 font-medium">Catégorie</th>
                    <th className="pb-3 font-medium text-center">Stock en Réserve</th>
                    <th className="pb-3 font-medium">Unité</th>
                  </tr>
                </thead>
                <tbody className="divide-y">
                  {items?.length === 0 ? (
                    <tr>
                      <td colSpan={4} className="py-6 text-center text-gray-500 italic">
                        Aucun article dans la réserve pour le moment.
                      </td>
                    </tr>
                  ) : (
                    items?.map((item: any) => (
                      <tr key={item.id} className="hover:bg-gray-50">
                        <td className="py-3 font-medium text-gray-900">{item.name}</td>
                        <td className="py-3 text-gray-600">{item.category}</td>
                        <td className="py-3 text-center">
                          <span className={`inline-block px-2.5 py-1 rounded-full text-xs font-semibold ${item.current_stock > 0 ? 'bg-green-100 text-green-800' : 'bg-red-100 text-red-800'}`}>
                            {item.current_stock}
                          </span>
                        </td>
                        <td className="py-3 text-gray-600">{item.unit}</td>
                      </tr>
                    ))
                  )}
                </tbody>
              </table>
            </div>
          </CardContent>
        </Card>
      </main>
    </div>
  );
}
